// Independent carryless convolution/long division uses a uint64 product and no
// encoder multiplication or repeated shift/reduce field routines.
func binaryOracleMul(a, b uint32, s int, p uint32) uint32 {
	var product uint64
	for i := 0; i < s; i++ {
		for j := 0; j < s; j++ {
			product ^= uint64((a>>uint(i))&(b>>uint(j))&1) << uint(i+j)
		}
	}
	for i := 2*s - 2; i >= s; i-- {
		if product&(uint64(1)<<uint(i)) != 0 {
			product ^= uint64(p) << uint(i-s)
		}
	}
	return uint32(product)
}
func quadraticOracleMul(f QuadraticField, x, y QuadraticElement) QuadraticElement {
	mul := func(a, b uint32) uint32 { return binaryOracleMul(a, b, f.Base().Degree(), f.Base().Polynomial()) }
	return QuadraticElement{A: mul(x.A, y.A) ^ mul(f.Delta(), mul(x.B, y.B)), B: mul(x.A, y.B) ^ mul(x.B, y.A) ^ mul(x.B, y.B)}
}
func TestTableFreeFields(t *testing.T) {
	random := rand.New(rand.NewSource(9088))
	for s := 2; s <= 20; s++ {
		p, err := BinaryPolynomial(s)
		if err != nil {
			t.Fatal(err)
		}
		f, err := NewBinaryField(s, p)
		if err != nil {
			t.Fatalf("s=%d: %v", s, err)
		}
		q := uint32(1) << uint(s)
		if s <= 8 {
			for a := uint32(0); a < q; a++ {
				for b := uint32(0); b < q; b++ {
					if f.Mul(a, b) != binaryOracleMul(a, b, s, p) {
						t.Fatal("exhaustive base mismatch", s, a, b)
					}
				}
			}
		}
		ext, err := NewQuadraticField(f)
		if err != nil {
			t.Fatal(err)
		}
		if f.Trace(ext.Delta()) != 1 {
			t.Fatal("invalid tower")
		}
		for i := 0; i < 1000; i++ {
			a, b := uint32(random.Intn(int(q))), uint32(random.Intn(int(q)))
			if f.Mul(a, b) != binaryOracleMul(a, b, s, p) || f.Pow(a, uint64(q)) != a {
				t.Fatal("base identity mismatch", s)
			}
			if a != 0 {
				inv, err := f.Inverse(a)
				if err != nil || f.Mul(a, inv) != 1 {
					t.Fatal("base inverse mismatch", s)
				}
			}
			x, y := QuadraticElement{A: a, B: b}, QuadraticElement{A: uint32(random.Intn(int(q))), B: uint32(random.Intn(int(q)))}
			if ext.Mul(x, y) != quadraticOracleMul(ext, x, y) {
				t.Fatal("tower multiplication mismatch", s)
			}
			// q-Frobenius conjugates u to u+1, and applying it twice is identity.
			if ext.Pow(x, uint64(q)) != (QuadraticElement{A: a ^ b, B: b}) || ext.Pow(x, uint64(q)*uint64(q)) != x {
				t.Fatal("tower Frobenius mismatch", s)
			}
			if a != 0 || b != 0 {
				inv, err := ext.Inverse(x)
				if err != nil || ext.Mul(x, inv) != (QuadraticElement{A: 1}) {
					t.Fatal("tower inverse mismatch", s)
				}
			}
		}
		if _, err := f.Inverse(0); err == nil {
			t.Fatal("zero inverse accepted")
		}
		if _, err := ext.Inverse(QuadraticElement{}); err == nil {
			t.Fatal("zero tower inverse accepted")
		}
		if _, err := ext.Inverse(QuadraticElement{A: q}); err == nil {
			t.Fatal("noncanonical tower element")
		}
	}
	// Both descriptions contain only scalars; sizes are independent of s and q.
	if unsafe.Sizeof(BinaryField{}) > 16 || unsafe.Sizeof(QuadraticField{}) > 32 {
		t.Fatal("arithmetic descriptions grew unexpectedly")
	}
}
func TestQuadraticExhaustiveTiny(t *testing.T) {
	for s := 2; s <= 4; s++ {
		p, _ := BinaryPolynomial(s)
		base, _ := NewBinaryField(s, p)
		f, _ := NewQuadraticField(base)
		q := uint32(1) << uint(s)
		for a := uint32(0); a < q*q; a++ {
			for b := uint32(0); b < q*q; b++ {
				x, y := QuadraticElement{A: a & (q - 1), B: a >> uint(s)}, QuadraticElement{A: b & (q - 1), B: b >> uint(s)}
				if f.Mul(x, y) != quadraticOracleMul(f, x, y) {
					t.Fatal("exhaustive extension mismatch", s, a, b)
				}
			}
		}
	}
}
func TestBinaryFieldValidation(t *testing.T) {
	for _, s := range []int{-1, 0, 1, 21, 100} {
		if _, err := BinaryPolynomial(s); err == nil {
			t.Fatal("invalid degree accepted")
		}
	}
	for _, p := range []uint32{0, 0x10, 0x11, 0x15, 0x27} {
		if _, err := NewBinaryField(4, p); err == nil {
			t.Fatal("invalid polynomial accepted", p)
		}
	}
	if _, err := NewQuadraticField(BinaryField{}); err == nil {
		t.Fatal("zero field accepted")
	}
}

// Existing checked-in version-1 artifacts are a compatibility fixture. The
// expected hashes were measured before table removal, not regenerated by this test.
func TestTableFreeLegacyOutputs(t *testing.T) {
	root := "../../experiments/encoder-comparison"
	raw, err := os.ReadFile(filepath.Join(root, "results.json"))
	if err != nil {
		t.Fatal(err)
	}
	var report struct {
		Scenarios []struct {
			Preprocessing struct{ S, T int }
			Methods       []struct {
				Method string `json:"method"`
				Output string `json:"output_sha256"`
			}
		}
	}
	if err := json.Unmarshal(raw, &report); err != nil {
		t.Fatal(err)
	}
	for _, scenario := range report.Scenarios {
		stem := filepath.Join(root, "artifacts", fmt.Sprintf("s%02d-m2-t%d", scenario.Preprocessing.S, scenario.Preprocessing.T))
		for _, method := range scenario.Methods {
			if method.Method == "cyclic" {
				continue
			} // FFT and paper cover persisted tables/constants; Karatsuba is tested independently.
			data, err := os.ReadFile(stem + "-" + method.Method + ".json")
			if err != nil {
				t.Fatal(err)
			}
			var basis *OrbitBasis
			var encode func([]uint32) ([]uint32, error)
			if method.Method == "paper" {
				e, err := LoadPaperEncoder(data)
				if err != nil {
					t.Fatal(err)
				}
				basis, encode = e.Basis(), e.Encode
			} else {
				e, err := LoadCyclicEncoder(data)
				if err != nil {
					t.Fatal(err)
				}
				basis, encode = e.Basis(), e.Encode
			}
			r := rand.New(rand.NewSource(71))
			message := make([]uint32, basis.Dimension())
			for i := range message {
				message[i] = uint32(r.Intn(int(basis.base.q)))
			}
			word, err := encode(message)
			if err != nil {
				t.Fatal(err)
			}
			bytes := make([]byte, 4*len(word))
			for i, x := range word {
				binary.LittleEndian.PutUint32(bytes[4*i:], x)
			}
			digest := sha256.Sum256(bytes)
			if hex.EncodeToString(digest[:]) != method.Output {
				t.Fatal("legacy codeword changed", stem, method.Method)
			}
		}
	}
}

// The first larger code is checked against actual degree-t curves using an
// independent base-field arithmetic path, not just two encoders sharing setup.
func TestS9StoredArtifactCurveConstraints(t *testing.T) {
	data, err := os.ReadFile("../../experiments/tablefree-encoding/artifacts/s09-m2-t7-cyclic-fft.json")
	if err != nil {
		t.Fatal(err)
	}
	encoder, err := LoadCyclicEncoder(data)
	if err != nil {
		t.Fatal(err)
	}
	basis := encoder.Basis()
	if basis.Dimension() != 17200 || basis.Length() != 262144 {
		t.Fatal("unexpected exact s=9 dimensions")
	}
	r := rand.New(rand.NewSource(7009))
	message := make([]uint32, basis.Dimension())
	for i := range message {
		message[i] = uint32(r.Intn(512))
	}
	word, err := encoder.Encode(message)
	if err != nil {
		t.Fatal(err)
	}
	for j, x := range message {
		position, err := basis.SystematicPosition(j)
		if err != nil || word[position] != x {
			t.Fatal("s=9 information-set mismatch")
		}
	}
	for trial := 0; trial < 8; trial++ {
		var curve [2][8]uint32
		for axis := range curve {
			for j := range curve[axis] {
				curve[axis][j] = uint32(r.Intn(512))
			}
		}
		var sum uint32
		for z := uint32(0); z < 512; z++ {
			var point [2]uint32
			for axis := range curve {
				for j := 7; j >= 0; j-- {
					point[axis] = binaryOracleMul(point[axis], z, 9, 0x211) ^ curve[axis][j]
				}
			}
			sum ^= word[point[0]+512*point[1]]
		}
		if sum != 0 {
			t.Fatal("s=9 codeword violates sampled degree-7 curve", trial)
		}
	}
}

func TestTableFreePreparedArtifactsStayIdentical(t *testing.T) {
	root := "../../experiments/encoder-comparison/artifacts/"
	basisJSON, err := os.ReadFile(root + "s08-m2-t8-basis.json")
	if err != nil {
		t.Fatal(err)
	}
	basis, err := LoadOrbit(basisJSON)
	if err != nil {
		t.Fatal(err)
	}
	paper, err := basis.PreparePaper()
	if err != nil {
		t.Fatal(err)
	}
	cyclic, err := basis.PrepareCyclicFFT()
	if err != nil {
		t.Fatal(err)
	}
	for name, marshal := range map[string]func() ([]byte, error){"paper": paper.MarshalBinary, "cyclic-fft": cyclic.MarshalBinary} {
		original, err := os.ReadFile(root + "s08-m2-t8-" + name + ".json")
		if err != nil {
			t.Fatal(err)
		}
		got, err := marshal()
		if err != nil || !bytes.Equal(got, original) {
			t.Fatal("prepared artifact bytes changed", name, err)
		}
	}
}
