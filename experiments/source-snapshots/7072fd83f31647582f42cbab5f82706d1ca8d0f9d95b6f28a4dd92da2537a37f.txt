#ifndef LIFTRS_C_API_H
#define LIFTRS_C_API_H
#include <stddef.h>
#include <stdint.h>
#ifdef __cplusplus
extern "C" {
#endif
/* Immutable C++-owned plan. Create copies the bounded, pointer-free Go plan
 * description. Apply borrows an existing uint32 buffer only until return; there
 * is no per-element boundary call, retained Go pointer, or input/output copy.
 * Versions 1/2 store 32-bit domain counts; 3/4 insert a high count word after
 * the low word. Odd versions use a canonical basis, even versions an explicit
 * basis. Only counts widen: all field values and basis points remain uint32. */
typedef struct liftrs_plan liftrs_plan;
int liftrs_plan_create(const uint32_t* words, size_t count, liftrs_plan** out,
                       char* error, size_t error_capacity);
int liftrs_plan_apply(const liftrs_plan* plan, uint32_t* data, size_t count,
                      int inverse, uint64_t* kernel_ns,
                      char* error, size_t error_capacity);
/* General-basis affine execution; start is a packed field point, not a row index. */
int liftrs_plan_apply_at(const liftrs_plan*, uint32_t*, size_t, uint32_t start,
                        int inverse, uint64_t*, char*, size_t);
/* In-memory matrix decomposition. scratch must be disjoint and count elements.
 * rows includes S_split; columns uses S_split(high basis). Both are immutable. */
int liftrs_matrix_apply(const liftrs_plan* rows, const liftrs_plan* columns,
                        uint32_t* data, uint32_t* scratch, size_t count,
                        unsigned split, unsigned workers, int inverse, uint64_t* stage_ns,
                        char* error, size_t error_capacity);
/* One-time code-only preparation; caller must exclude concurrent execution. */
int liftrs_plan_prepare_taylor(liftrs_plan*, unsigned high_levels, char*, size_t);
/* Complete cyclic online path. Constants and caller buffers are borrowed
 * only until return. arena has max(2*l,n) elements; h has k, scratch/word n.
 * word may exactly equal scratch: assembly follows the final transform. All
 * other overlap, including partial scratch/word overlap, is forbidden.
 * stats[6]: Taylor, columns/transposes, rows, buffer setup, products, assembly.
 * The Go binding checks all forbidden overlap before entering this function. */
int liftrs_online_encode(const liftrs_plan*, const liftrs_plan*, const liftrs_plan*,
    const liftrs_plan*, unsigned, unsigned, unsigned, uint32_t,
    const uint32_t*, const uint32_t*, const uint32_t*, size_t, size_t, size_t,
    uint32_t*, size_t, uint32_t*, uint32_t*, uint32_t*, uint64_t*, char*, size_t);
/* Explicit compact experiment: bounded columns, no h storage, word==arena,
 * 2L<=N, uint16 compaction/disjoint scatter/backwards expansion. scratch_count
 * covers min(workers,n1/min(64,n1))*min(64,n1)*n2 for each domain.
 * stats[8] extend normal stages with compaction/move and expansion nanoseconds. */
int liftrs_online_encode_compact(size_t scratch_count,const liftrs_plan*,const liftrs_plan*,const liftrs_plan*,const liftrs_plan*,unsigned,unsigned,unsigned,uint32_t,const uint32_t*,const uint32_t*,const uint32_t*,size_t,size_t,size_t,uint32_t*,size_t,uint32_t*,uint32_t*,uint32_t*,uint64_t*,char*,size_t);
/* File-backed fast experiment. Immutable verified little-endian code is read
 * through a borrowed descriptor; no pointer/descriptor is retained. Requires
 * 2L<=N, disjoint arena/scratch, each N elements. Scratch is bounded read tile
 * between forward/inverse and owning output after inverse. tile<=min(N,2^26).
 * stats[11]: normal6, two zero compact slots, read/range-check ns, bytes, chunks.
 * Read time for prefix P overlaps buffer time; timings must not be double-counted. */
int liftrs_online_encode_file(int,uint64_t,uint64_t,size_t,const liftrs_plan*,const liftrs_plan*,const liftrs_plan*,const liftrs_plan*,unsigned,unsigned,unsigned,uint32_t,const uint32_t*,size_t,size_t,size_t,uint32_t*,uint32_t*,uint64_t*,char*,size_t);
/* Diagnostic nanoseconds: stage steady_clock, system wall, POSIX monotonic,
 * process CPU (all threads). No encoding state or buffers are changed. */
int liftrs_read_clocks(uint64_t* values, char* error, size_t capacity);
size_t liftrs_plan_owned_bytes(const liftrs_plan* plan);
void liftrs_plan_destroy(liftrs_plan* plan);
#ifdef __cplusplus
}
#endif
#endif
