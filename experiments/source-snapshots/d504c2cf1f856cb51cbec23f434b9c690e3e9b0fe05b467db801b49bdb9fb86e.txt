def link_inputs(destination,reference):
    destination.mkdir(parents=True,exist_ok=False)
    for name in ('online.bin','online-metadata.json'):(destination/name).symlink_to(reference/name)

def verify_phases(out,s,streamed=False):
    phases=json.loads((out/'phases.json').read_text());p={row['phase']:row for row in phases};n,k=p['load-online']['n'],p['load-online']['k']
    assert n==1<<(2*s) and 0<k<=n//2
    canonical=None;encodings=[]
    for repeat in (0,1):
        check=p[f'verify-{repeat}'];assert check['systematic_checked']==k and check['curves_checked']==16 and check['repeat_match']
        if canonical is None:canonical=check['canonical_sha256']
        assert check['canonical_sha256']==canonical
        enc=p[f'encode-{repeat}'];stats=enc['cuda_stages'];assert stats['native_total_ns']<=stats['elapsed_ns']+1_000_000
        assert stats['owned_device_bytes']<=(512<<20 if s<=10 else 4<<30 if s==14 else (40 if streamed else 56)<<30)
        item={'seconds':enc['elapsed_ns']/1e9,'cuda_stages':stats}
        if streamed:
            tile=min(n,1<<26)
            assert enc['code_upload_bytes']==4*(n+k) and enc['spectrum_chunks']==(n+tile-1)//tile and enc['stream_tile_bytes']==4*tile
            assert 0<=enc['code_upload_seconds']<=stats['host_to_device_ns']/1e9+0.001
            item.update({key:enc[key] for key in ('code_upload_seconds','code_upload_bytes','spectrum_chunks','stream_tile_bytes')})
        encodings.append(item)
    assert p['pack-output']['canonical_sha256']==canonical and p['pack-output']['packed_bytes']==(n*s+7)//8
    assert p['verify-packed']['positions_checked']==1027 and p['verify-packed']['readback_match']
    return {'s':s,'n':n,'k':k,'canonical_sha256':canonical,'packed_sha256':p['pack-output']['sha256'],'encodings':encodings,'phase_seconds':{row['phase']:row['elapsed_ns']/1e9 for row in phases}}

def validate_metadata(meta,size):
    assert meta['s']==16 and meta['t']==5 and meta['k']==918672421 and meta['n']==1<<32
    assert meta['identity_verified'] and meta['sha256']==EXPECTED and meta['bytes']==size==20854558905


def verify_small(out,streamed):
    baseline=json.loads((out/'baseline/online-checks.json').read_text())
    candidate=json.loads((out/'candidate/online-checks.json').read_text())
    by={r['s']:r for r in candidate};assert sorted(by)==[2,3,4,6,8]
    for row in baseline:
        other=by[row['s']]
        if row['s'] in (2,3,4):assert row['unsupported_prefix'] and other['unsupported_prefix'];continue
        assert row['matches']==other['matches']==4 and row['closed'] and other['closed'] and row['invalid_messages'] and other['invalid_messages']
        expected=4*(row['n']+row['k']+1) if streamed else 0
        assert row['owned_bytes']-other['owned_bytes']==expected
    memcheck=json.loads((out/'memcheck/online-checks.json').read_text())
    assert [(r['s'],r.get('matches'),r.get('unsupported_prefix')) for r in candidate]==[(r['s'],r.get('matches'),r.get('unsupported_prefix')) for r in memcheck]
    return {'word_cases':2,'prefix_rejections':3,'memcheck':True,'streamed':streamed}

def verify_packed(out,s,mode,streamed):
    phases=json.loads((out/'phases.json').read_text());p={row['phase']:row for row in phases};n,k=p['load-online']['n'],p['load-online']['k']
    assert n==1<<(2*s) and 0<k<=n//2
    canonical=None;packed=None;encodings=[]
    for repeat in (0,1):
        check=p[f'verify-{repeat}'];assert check['systematic_checked']==k and check['curves_checked']==16 and check['repeat_match']
        read=p[f'verify-packed-{repeat}'];assert read['positions_checked']==1027 and read['readback_match']
        enc=p[f'encode-output-{repeat}'];st=enc['packed_stats']
        assert st['packed_bytes']==n*s//8 and st['native_total_ns']<=enc['pipeline_ns']+1_000_000
        assert enc['output_mode']==mode and enc['host_canonical_array_bytes']==(0 if mode=='packed' else 4*n)
        assert 0<st['host_buffer_bytes']<=8<<20 and st['owned_device_bytes']<=(512<<20 if s<=10 else 4<<30 if s==14 else (40 if streamed else 56)<<30)
        if canonical is None:canonical=check['canonical_sha256'];packed=st['packed_sha256']
        assert check['canonical_sha256']==canonical and st['packed_sha256']==packed==read['packed_sha256']
        tile=min(n,1<<26)
        assert enc['code_upload_bytes']==(4*(n+k) if streamed else 0) and enc['spectrum_chunks']==((n+tile-1)//tile if streamed else 0)
        if mode=='packed':assert st['elapsed_ns']>=st['native_total_ns'] and enc['cpu_pack_write_ns']==0
        encodings.append({'seconds':enc['pipeline_ns']/1e9,'phase_seconds':enc['elapsed_ns']/1e9,'packed_stats':st,'cpu_pack_write_ns':enc['cpu_pack_write_ns'],'sync_ns':enc['sync_ns']})
    return {'s':s,'n':n,'k':k,'output_mode':mode,'canonical_sha256':canonical,'packed_sha256':packed,'encodings':encodings,'phase_seconds':{row['phase']:row['elapsed_ns']/1e9 for row in phases}}

def verify_memory(record,baseline,streamed):
    old=baseline['encoding']['encodings'][0]['cuda_stages']['owned_device_bytes']
    n,k=record['encoding']['n'],record['encoding']['k'];saved=4*(n+k+1) if streamed else 0;expected=old-saved
    for enc in record['encoding']['encodings']:assert enc['packed_stats']['owned_device_bytes']==expected
    return {'resident_reuse_owned_bytes':old,'candidate_owned_bytes':expected,'saved_device_bytes':saved,'streamed':streamed,'comparison':'same fast layouts/tables/plans/parity; historical full reuse timing, matched s14 resident/streamed'}
